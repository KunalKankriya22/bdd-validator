**Status**: ENABLED

```plsql

	PRODUCT_TYPE,
	TIER,
	ALTERNATE_TIER_FLAG,
	EFF_DATE,
	TERM_DATE,

	PROGRAM_ID,
	AUDIT_INSERT_USER,
	AUDIT_INSERT_DATE
  )
  values
  (	:new.CHAPTER_ID,
	:new.CUSTOMER_ID,
	:new.PRODUCT_TYPE,
	:new.TIER,
	:new.ALTERNATE_TIER_FLAG,
	:new.EFF_DATE,
	:new.TERM_DATE,
	:new.LAST_MODIFIED_DATE,
	:new.AUDIT_KEY,
	'UPDATE',LT_USER_SESSION.GET_USERID,
    sysdate
  ) ;
  end ;
```
