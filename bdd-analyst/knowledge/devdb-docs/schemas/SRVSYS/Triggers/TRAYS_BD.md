# TRAYS_BD

	:old.WAREHOUSE_ID,
	:old.DEVICE_TYPE,
	:old.DEVICE_NUMBER,
	:old.RACK_NUMBER,
	:old.SHELF_NUMBER,
	:old.TRAY_NUMBER,
	:old.TRAY_DESCRIPTION,
	:old.PRODUCT_TYPE,
	:old.BLOOD_TYPE,
	:old.ITEM_COUNT,
	:old.TRAY_STATUS,
