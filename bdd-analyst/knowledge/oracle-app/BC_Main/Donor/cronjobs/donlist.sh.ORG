#!/bin/ksh
# This shell is started by cron process oracle.ctab to automatically
# Run the Enlistment server
#

PATH=$PATH:/usr/local/bin
for ORACLE_SID in qtrak
do
export ORACLE_SID
ORAENV_ASK=NO
. oraenv
cd /home/donsys/qtrak/donorexec
sqlplus donsys/donsys @donlist.sql
if [ $? -eq 0 ]
then
     COMPL="completed"
else
     COMPL="aborted"
fi;
done
